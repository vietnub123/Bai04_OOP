cmake_minimum_required(VERSION 3.16)
project(PayrollLab04 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(payroll payroll.cpp)
target_include_directories(payroll PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(payroll PUBLIC /W4 /utf-8)
else()
  target_compile_options(payroll PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(payroll_demo main.cpp)
target_link_libraries(payroll_demo PRIVATE payroll)
add_executable(payroll_tests tests.cpp)
target_link_libraries(payroll_tests PRIVATE payroll)
enable_testing()
add_test(NAME payroll_tests COMMAND payroll_tests)
